Prompt : Générer un Dockerfile qui conteneurise l'application ci-dessus. L'application ci-dessus s'appelle api_app.py.    #1

FROM python:3.11-slim    #2

ENV PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1 \
    PORT=8000

WORKDIR /app

RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential curl ca-certificates && \
    rm -rf /var/lib/apt/lists/*    #3

COPY requirements.txt /app/requirements.txt    #4
RUN if [ -f /app/requirements.txt ]; then \
    pip install --no-cache-dir -r /app/requirements.txt; \
    else \
    pip install --no-cache-dir "uvicorn[standard]" fastapi; \
fi

COPY . /app

EXPOSE ${PORT}    #5
CMD ["sh", "-c", "uvicorn 02_app:app --host 0.0.0.0 --port ${PORT}"]

#1 L'instruction utilisateur pour générer le Dockerfile, précisant que l'application à conteneuriser est api_app.py.
#2 Utilise une image de base Python 3.11 légère (slim) pour réduire la taille du conteneur.
#3 Installe les dépendances système nécessaires (build-essential, curl, ca-certificates), puis nettoie le cache apt pour réduire la taille de l'image.
#4 Copie le fichier requirements.txt dans le conteneur. S'il existe, installe les dépendances Python ; sinon, installe uvicorn[standard] et fastapi par défaut.
#5 Expose le port défini par la variable d'environnement PORT (8000 par défaut) et lance l'application avec uvicorn.
